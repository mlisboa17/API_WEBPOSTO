    """Cache-Aside Strategy com Valkey/Redis 7.0"""
    
    async def get_centros_custo_cached(self) -> List[Dict]:
        """
        1. Try cache (key: "centros_custo:all")
        2. If miss → API call
        3. Populate cache com TTL 3600s
        """
        cache_key = "centros_custo:all"
        
        # Try cache
        cached = await self.redis.get(cache_key)
        if cached:
            return json.loads(cached)
        
        # Cache miss
        data = await self.client.discover_centros_custo()
        
        # Populate cache
        await self.redis.setex(
            cache_key,
            3600,  # TTL: 1 hora
            json.dumps(data)
        )
        
        return data
```

from dataclasses import dataclass
from typing import List
from decimal import Decimal
from datetime import datetime, timezone

@dataclass(frozen=True)
class Lancamento:
    """Agregado: Lançamento com múltiplos rateios por CC"""
    
    id: 'LancamentoID'
    data: datetime
    valor_total: 'ValorMonetario'
    descricao: str
    rateios: List['RateioCentroCusto']
    usuario: str
    data_criacao: datetime
    
    def validar_rateios(self) -> bool:
        """Garantir que soma dos rateios == valor_total"""
        soma = sum(r.valor.amount for r in self.rateios)
        return soma == self.valor_total.amount
    
    def validar_percentuais(self) -> bool:
        """Percentuais devem somar 100%"""
        total = sum(r.percentual for r in self.rateios)
        return total == Decimal("100")
    
    @classmethod
    def criar(cls, id: str, data: datetime, valor: float,
              descricao: str, rateios: List, usuario: str) -> 'Lancamento':
        """Factory com validações"""
        lancamento = cls(
            id=LancamentoID(id),
            data=data,
            valor_total=ValorMonetario(valor),
            descricao=descricao,
            rateios=[RateioCentroCusto.criar(r) for r in rateios],
            usuario=usuario,
            data_criacao=datetime.now(timezone.utc)
        )
        
        # Validar
        if not lancamento.validar_rateios():
            raise RateioDivergenceError(
                f"Soma dos rateios != {valor}"
            )
        
        if not lancamento.validar_percentuais():
            raise PercentualInvalidoError(
                "Percentuais não somam 100%"
            )
        
        return lancamento
```

Eventos de Domínio:
- LancamentoCriado
- RateioAdicionado
- DivergenciaDetectada

class CentroCustoID:

        if not self.valor or len(self.valor) > 50:
            raise ValueError("CC ID inválido")

@dataclass(frozen=True)
class ValorMonetario:
    """Value Object para valores"""
    amount: Decimal
    currency: str = "BRL"
    
    def __post_init__(self):
        if self.amount < 0:
            raise ValueError("Valor não pode ser negativo")

@dataclass(frozen=True)
class RateioCentroCusto:
    """Rateio para um Centro de Custo"""
    centro_custo_id: CentroCustoID
    valor: ValorMonetario
    percentual: Decimal  # 0-100

    def criar(cls, cc_id: str, valor: float, pct: float):
        return cls(
            centro_custo_id=CentroCustoID(cc_id),
            valor=ValorMonetario(Decimal(str(valor))),

import hashlib
import json

class HashConsistencyEngine:
    """Gera hash para validar integridade de rateios"""
    
    @staticmethod
    def gerar_hash_lancamento(lancamento: Dict) -> str:
        """
        SHA-256 do payload original vs processado
        
        Campos inclusos: id, data, valor_total, rateios
        Campos exclusos: timestamp, usuario
        """
        payload = {
            "id": lancamento["id"],
            "data": lancamento["data"],
            "valor_total": lancamento["valor_total"],
            "rateios": sorted([
                (r["cc_id"], float(r["valor"]))
                for r in lancamento["rateios"]
            ])
        }
        
        json_str = json.dumps(payload, sort_keys=True)
        return hashlib.sha256(json_str.encode()).hexdigest()
    
    @staticmethod
    def validar_integridade(hash_original: str,
                            hash_processado: str) -> bool:
        """Retorna True se hashes coincidem"""
        return hash_original == hash_processado
```

class AnomalyDetector:
    DESVIO_MAXIMO_PCT = 15  # 15%
    
    def detectar_lancamento_sem_cc(self, lancamento: Lancamento) -> bool:
        """
        Detecta se lançamento não tem Centro de Custo
        Retorna True se anomalia (sem CC)
        """
        if not lancamento.rateios:
            return True
        
        if len(lancamento.rateios) == 1 and \
           lancamento.rateios[0].cc_id == "CC_GERAL":
            return True
        
        return False
    
    def detectar_desvio_operacional(self,
                                    rateios: List,
                                    cc_tipo: str) -> bool:
        """
        Detecta desvios >15% em CCs operacionais
        """
        if cc_tipo not in ["Operacional", "Pista"]:
            return False
        
        total = sum(r.valor for r in rateios)
        desvio_pct = abs(total - VALOR_ESPERADO) / VALOR_ESPERADO * 100

class AuditEngine:
    """Registro completo de auditoria"""
    
    def __init__(self, db: MongoClient, event_bus: EventBus):
        self.audit_collection = db["auditoria"]
        self.event_bus = event_bus
    
    async def registrar_operacao(self,
                                 operacao: str,
                                 valor_antes: Dict,
                                 valor_depois: Dict,
                                 usuario: str,
                                 motivo: str,
                                 ip_origem: str) -> None:
        """
        Registro completo com integridade
        
        Campos:
        - operacao: "criar_lancamento", "atualizar_rateio"
        - valor_antes: Estado anterior
        - valor_depois: Estado novo
        - hash_antes: SHA256 antes
        - hash_depois: SHA256 depois
        - usuario: Quem fez
        - motivo: Por quê
        - ip_origem: IP
        - timestamp: Data/hora
        - divergencia: True/False
        """
        
        hash_antes = HashConsistencyEngine.gerar_hash(valor_antes)
        hash_depois = HashConsistencyEngine.gerar_hash(valor_depois)
        
        registro = {
            "operacao": operacao,
            "valor_antes": valor_antes,
            "valor_depois": valor_depois,
            "hash_antes": hash_antes,
            "hash_depois": hash_depois,
            "usuario": usuario,
            "motivo": motivo,
            "ip_origem": ip_origem,
            "timestamp": datetime.now(timezone.utc),
            "divergencia": hash_antes != hash_depois
        }
        
        await self.audit_collection.insert_one(registro)
        
        # Publicar para análise
        await self.event_bus.publicar(
            OperacaoAuditada(
                operacao=operacao,
                divergencia=registro["divergencia"]
            )
        )

class CacheAsideStrategy:
    """Estratégia para reduzir latência em 90%"""
    
    def __init__(self, cache: Redis, db: MongoClient):
        self.cache = cache
        self.db = db
    
    async def obter_lancamento(self, lancamento_id: str):
        """
        1. Try cache
        2. Se miss → Query BD
        3. Populate cache (TTL: 3600s)
        """
        
        cache_key = f"lancamento:{lancamento_id}"
        
        # 1. Cache
        cached = await self.cache.get(cache_key)
        if cached:
            return json.loads(cached)
        
        # 2. BD
        doc = await self.db["lancamentos"].find_one(
            {"_id": lancamento_id}
        )
        
        if not doc:
            raise NotFound()
        
        # 3. Populate
        await self.cache.setex(
            cache_key,
            3600,  # TTL
            json.dumps(doc, default=str)
        )
        
        return doc
```
