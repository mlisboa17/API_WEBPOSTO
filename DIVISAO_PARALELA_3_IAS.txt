1️⃣ ENTITY: LANCAMENTO (Agregado Raiz)
   ──────────────────────────────────

   ```python
   @dataclass(frozen=True)
   class Lancamento:
       id: LancamentoID
       data: datetime
       valor_total: ValorMonetario
       descricao: str
       rateios: List[RateioCentroCusto]
       usuario: str
       data_criacao: datetime
       
       def validar_rateios(self) -> bool:
           """Garantir que soma dos rateios == valor_total"""
           soma = sum(r.valor.amount for r in self.rateios)
           return soma == self.valor_total.amount
       
       @classmethod
       def criar(cls, id: str, data: datetime, valor: float, 
                 descricao: str, rateios: List, usuario: str):
           lancamento = cls(
               id=LancamentoID(id),
               data=data,
               valor_total=ValorMonetario(valor),
               descricao=descricao,
               rateios=[RateioCentroCusto.criar(...) for r in rateios],
               usuario=usuario,
               data_criacao=datetime.now(timezone.utc)
           )
           
           if not lancamento.validar_rateios():
               raise RateioDivergenceError(
                   f"Soma dos rateios ({soma}) != {valor}"
               )
           
           return lancamento
   ```
   
   Eventos de Domínio:
   - LancamentoCriado

2️⃣ VALUE OBJECTS

   ```python
   @dataclass(frozen=True)
   class CentroCustoID:
       valor: str
       
       def __post_init__(self):
           if not self.valor or len(self.valor) > 50:
               raise ValueError("CC ID inválido")
   
   @dataclass(frozen=True)
   class ValorMonetario:
       amount: Decimal
       currency: str = "BRL"
       
       def __post_init__(self):
           if self.amount < 0:
               raise ValueError("Valor não pode ser negativo")
   
   @dataclass(frozen=True)
   class RateioCentroCusto:
       centro_custo_id: CentroCustoID
       valor: ValorMonetario
       percentual: Decimal  # 0-100
       
       @classmethod
       def criar(cls, cc_id: str, valor: float, percentual: float):
           return cls(
               centro_custo_id=CentroCustoID(cc_id),
               valor=ValorMonetario(Decimal(str(valor))),
               percentual=Decimal(str(percentual))
           )
   ```
   
   Validações:
   - ✓ Imutabilidade (frozen=True)
   - ✓ Validação pós-criação (__post_init__)
   - ✓ Igualdade por valor (dataclass)

1️⃣ HASH CONSISTENCY ENGINE
   ────────────────────────
   Implementar em: src/algorithms/hash_consistency.py
   
   ```python
   import hashlib
   from typing import Any, Dict
   
   class HashConsistencyEngine:
       @staticmethod
       def gerar_hash_lancamento(lancamento: Dict[str, Any]) -> str:
           """
           Gera hash SHA-256 do lançamento original vs processado
           
           Campos inclusos: id, data, valor_total, rateios
           Campos exclusos: timestamp, usuario (mudam)
           """
           payload = {
               "id": lancamento["id"],
               "data": lancamento["data"],
               "valor_total": lancamento["valor_total"],
               "rateios": sorted([
                   (r["cc_id"], r["valor"]) 
                   for r in lancamento["rateios"]
               ])
           }
           
           json_str = json.dumps(payload, sort_keys=True)
           return hashlib.sha256(json_str.encode()).hexdigest()
       
       @staticmethod
       def validar_integridade(hash_original: str, 
                               hash_processado: str) -> bool:
           """Retorna True se hashes coincidem"""
           return hash_original == hash_processado
   ```
   
   Fluxo:
   1. Gerar hash do payload original da API Quality
   2. Processar com rateios (Claude)
   3. Gerar hash final
   4. Comparar hashes
   5. Se divergência → Registrar em auditoria

   ```python
   class AnomalyDetector:
       DESVIO_MAXIMO_PERCENTUAL = 15  # 15%
       
       def detectar_lancamento_sem_cc(self, lancamento: Lancamento) -> bool:
           """
           Detecta se lançamento não tem Centro de Custo mapeado
           Retorna True se anomalia detectada
           """
           if not lancamento.rateios:
               return True
           
           if len(lancamento.rateios) == 1 and \
              lancamento.rateios[0].cc_id == "CC_GERAL":
               return True
           
           return False
       
       def detectar_desvio_cc_operacional(self, 
                                          rateios: List[RateioCentroCusto],
                                          cc_tipo: str) -> bool:
           """
           Detecta desvios acima de 15% em CCs operacionais
           Ex: "Despesa de Pista" pode ter max 15% variação
           """
           if cc_tipo not in ["Operacional", "Pista"]:
               return False
           
           total_alocado = sum(r.valor.amount for r in rateios)
           desvio = abs(total_alocado - VALOR_ESPERADO) / VALOR_ESPERADO * 100
           
           return desvio > self.DESVIO_MAXIMO_PERCENTUAL
       
       def gerar_relatorio_anomalias(self, 
                                      lancamentos: List[Lancamento]) -> Dict:
           """Gera relatório consolidado de anomalias"""
           return {
               "total_lancamentos": len(lancamentos),
               "sem_cc": len([l for l in lancamentos 
                             if self.detectar_lancamento_sem_cc(l)]),
               "desvios_operacional": len([...]),
               "timestamp": datetime.now(timezone.utc)
           }
   ```

   ```python
   class AuditEngine:
       def __init__(self, db: MongoClient, event_bus: EventBus):
           self.audit_collection = db["auditoria"]
           self.event_bus = event_bus
       
       async def registrar_operacao(self, 
                                    operacao: str,
                                    valor_antes: Dict,
                                    valor_depois: Dict,
                                    usuario: str,
                                    motivo: str) -> None:
           """
           Registra operação completa com auditoria
           
           Campos obrigatórios:
           - operacao: "criar_lancamento", "atualizar_rateio", etc
           - valor_antes: Estado anterior
           - valor_depois: Estado novo
           - usuario: Quem fez
           - motivo: Por quê
           - hash_antes: SHA256 do valor_antes
           - hash_depois: SHA256 do valor_depois
           - ip_origem: IP da requisição
           - timestamp: Data/hora automática
           """
           
           hash_antes = HashConsistencyEngine.gerar_hash(valor_antes)
           hash_depois = HashConsistencyEngine.gerar_hash(valor_depois)
           
           registro = {
               "operacao": operacao,
               "valor_antes": valor_antes,
               "valor_depois": valor_depois,
               "hash_antes": hash_antes,
               "hash_depois": hash_depois,
               "usuario": usuario,
               "motivo": motivo,
               "ip_origem": contexto.ip_origem,
               "timestamp": datetime.now(timezone.utc),
               "divergencia": hash_antes != hash_depois
           }
           
           await self.audit_collection.insert_one(registro)
           
           # Publicar evento para análise
           await self.event_bus.publicar(
   ```

4️⃣ CACHE-ASIDE STRATEGY
   ──────────────────────
   Implementar em: src/caching/cache_aside_strategy.py
   
   ```python
   class CacheAsideStrategy:
       def __init__(self, cache: Redis, db: MongoClient):
           self.cache = cache
           self.db = db
       
       async def obter_lancamento(self, lancamento_id: str) -> Lancamento:
           """
           1. Tenta cache
           2. Se miss → Busca BD
           3. Popula cache com TTL
           
           Redução de latência: ~90% de cache hits
           """
           
           cache_key = f"lancamento:{lancamento_id}"
           
           # 1. Try cache
           cached_data = await self.cache.get(cache_key)
           if cached_data:
               return Lancamento.from_dict(json.loads(cached_data))
           
           # 2. Cache miss → Query DB
           lancamento = await self.db["lancamentos"].find_one(
               {"_id": lancamento_id}
           )
           
           if not lancamento:
               raise LancamentoNaoEncontrado()
           
           # 3. Populate cache (TTL: 3600s)
           await self.cache.setex(
               cache_key,
               3600,
               json.dumps(lancamento, default=str)
           )
           
           return Lancamento.from_dict(lancamento)
   ```

# 1. PREPARAR AMBIENTE
────────────────────────────────────────────────────────────────────────────────
cd Api_WebPosto

# Instalar dependências
pip install -r requirements.txt

# Copiar variáveis de ambiente
cp .env.example .env

# 2. SUBIR INFRAESTRUTURA (Docker)
────────────────────────────────────────────────────────────────────────────────
docker compose up -d

# Verificar serviços
docker compose ps

# Logs em tempo real
docker compose logs -f

# 3. VALIDAR SAÚDE
────────────────────────────────────────────────────────────────────────────────

# FastAPI
curl http://localhost:8000/health

# MongoDB
curl http://localhost:27017

# Redis
redis-cli ping

# Prometheus
curl http://localhost:9090/-/healthy

# Grafana (http://localhost:3000 — user: admin / pass: admin)

# 4. RODAR TESTES PARA CADA IA
────────────────────────────────────────────────────────────────────────────────

# GEMINI 2.0 — Testes de API & Cache
pytest tests/unit/test_api_discovery.py -v

# CLAUDE 3.7 — Testes de DDD & Rateio
pytest tests/unit/test_domain_rateio.py -v

# GROK 4 — Testes de Auditoria & Carga
